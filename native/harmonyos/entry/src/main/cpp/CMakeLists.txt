cmake_minimum_required(VERSION 3.16)
project(orca_native_ssh LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(THIRD_PARTY_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party)
if(NOT EXISTS ${THIRD_PARTY_DIR}/libssh2/CMakeLists.txt)
  message(FATAL_ERROR "Run python scripts/fetch-native.py before building SSH support")
endif()
set(DISABLE_PACKAGE_CONFIG_AND_INSTALL ON CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
add_subdirectory(${THIRD_PARTY_DIR}/mbedtls ${CMAKE_CURRENT_BINARY_DIR}/mbedtls EXCLUDE_FROM_ALL)
set(CRYPTO_BACKEND "mbedTLS" CACHE STRING "" FORCE)
set(MBEDTLS_INCLUDE_DIR ${THIRD_PARTY_DIR}/mbedtls/include CACHE PATH "" FORCE)
# A file path keeps libssh2's export set separate from the parent mbedTLS targets.
set(MBEDCRYPTO_LIBRARY ${CMAKE_CURRENT_BINARY_DIR}/mbedtls/library/libmbedcrypto.a CACHE FILEPATH "" FORCE)
set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_ZLIB_COMPRESSION OFF CACHE BOOL "" FORCE)
set(LIBSSH2_USE_PKGCONFIG OFF CACHE BOOL "" FORCE)
add_subdirectory(${THIRD_PARTY_DIR}/libssh2 ${CMAKE_CURRENT_BINARY_DIR}/libssh2 EXCLUDE_FROM_ALL)
add_library(orcassh SHARED ssh_tunnel.cpp)
target_include_directories(orcassh PRIVATE ${THIRD_PARTY_DIR}/libssh2/include)
target_compile_options(orcassh PRIVATE -fexceptions -Wall -Wextra)
target_link_libraries(orcassh PRIVATE libssh2_static mbedcrypto mbedx509 mbedtls libace_napi.z.so)
add_dependencies(orcassh mbedcrypto)
